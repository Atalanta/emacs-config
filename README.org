#+TITLE: SNS Emacs Config
#+AUTHOR: Stephen Nelson-Smith

This is my literate Emacs configuration. A single file holds both the prose explaining the config and the code that implements it.

All changes go in this file. Tangling produces the actual =.el= files Emacs loads.

* Instructions

Clone to =~/.emacs.d=:

#+BEGIN_EXAMPLE
cd $HOME
git clone git://github.com/Atalanta/emacs-config.git .emacs.d
#+END_EXAMPLE

Tangle the config:

#+begin_example
emacs --batch --load org --eval '(org-babel-tangle-file "README.org")'
#+end_example

First launch downloads and compiles packages, which takes a few minutes.

To change the config: edit this file, then run =org-babel-tangle= (=C-c C-v t=). Only blocks with a =:tangle= header are written out.

The first block written to each file starts with a =lexical-binding= cookie. Emacs reads the cookie only from a file's first line, and without it the file uses the older dynamic binding, which Emacs 31 warns about.

* Overview

Tangling =README.org= produces:

- =early-init.el=
- =init.el=
- =sns-emacs-modules/=
- =sns-lisp/=

* The ~early-init.el~ file

Emacs reads this file first. It runs before any packages load and sets a few UI defaults.

#+begin_src emacs-lisp :tangle "early-init.el"
;;; -*- lexical-binding: t -*-
  (setq inhibit-startup-message t)
  (tool-bar-mode -1)
  (scroll-bar-mode -1)
  (menu-bar-mode -1)
  (setq frame-inhibit-implied-resize t)
#+end_src

* The ~init.el~ file

Global settings and module imports.

** Support Nix
On machines where I use Nix, add the Nix profile to Emacs's =PATH=:

#+begin_src emacs-lisp :tangle "init.el"
;;; -*- lexical-binding: t -*-
(defun setup-nix-path ()
  "Add Nix profile to Emacs environment."
  (let ((nix-bin (expand-file-name ".nix-profile/bin" (getenv "HOME"))))
    (add-to-list 'exec-path nix-bin)
    (setenv "PATH" (concat nix-bin ":" (getenv "PATH")))))

(setup-nix-path)
#+end_src

** Trust the Zscaler proxy
On the work Mac, Zscaler re-signs HTTPS connections with its own root. Emacs checks certificates with GnuTLS against =gnutls-trustfiles=, and the only default file on macOS, =/etc/ssl/cert.pem=, lacks that root. Every HTTPS connection then raises a Network Security Manager prompt. A prompt from a Lisp thread deadlocks Emacs. home-manager builds =~/.pki/zscaler/ca-bundle.pem= from the public CAs, minus the compromised eMudhra roots, plus the Zscaler root. Use it when it exists, so the personal machine keeps the defaults.

#+begin_src emacs-lisp :tangle "init.el"
(let ((work-ca-bundle (expand-file-name "~/.pki/zscaler/ca-bundle.pem")))
  (when (file-readable-p work-ca-bundle)
    (setq gnutls-trustfiles (list work-ca-bundle))))
#+end_src

** Ensure shell environment variables are available

GUI Emacs on macOS doesn't inherit shell environment variables. =exec-path-from-shell= imports them so Emacs can find tools like Git. Only run it under a window system; on a Linux terminal Emacs the environment is already correct.

#+begin_src emacs-lisp :tangle "init.el"
(use-package exec-path-from-shell
  :ensure t
  :config
  (when (memq window-system '(mac ns x))
    (exec-path-from-shell-initialize)))
#+end_src

** Use zsh as the shell

Use zsh for shell buffers and vterm. Fall back to bash where zsh isn't installed.

#+begin_src emacs-lisp :tangle "init.el"
  (setq shell-file-name (or (executable-find "zsh")
                             (executable-find "bash")
                             "/bin/bash"))
  (setq explicit-shell-file-name shell-file-name)
  (with-eval-after-load 'vterm
    (setq vterm-shell shell-file-name))
#+end_src

** Simplify auto-save and backup features
Git makes most of Emacs's backup features redundant. Disable backup files and lockfiles. Keep auto-save (Emacs's crash-recovery cache) but write it to a dedicated directory rather than alongside the working copy.

#+begin_src emacs-lisp :tangle "init.el"
(setq make-backup-files nil)

(setq auto-save-default t)
(setq auto-save-timeout 20)
(setq auto-save-interval 200)
(make-directory "~/.emacs.d/auto-save/" t)
(setq auto-save-file-name-transforms '((".*" "~/.emacs.d/auto-save/" t)))

(setq create-lockfiles nil)
#+end_src

** Run as a server

Run an Emacs server so external tools (=emacsclient=, scripts, AI agents)
can evaluate elisp in this Emacs. Pin the socket to =~/.emacs.d/server/=
so its path doesn't depend on =$TMPDIR= and is reachable from any shell.
Create the directory with mode =700=, because =server.el= refuses to run if it's
group- or world-accessible, since a writable socket directory would let
other users spoof the server. Start the server only if one isn't already
running.

#+begin_src emacs-lisp :tangle "init.el"
  (require 'server)
  (setq server-socket-dir (expand-file-name "server/" user-emacs-directory))
  (with-file-modes #o700
    (make-directory server-socket-dir t))
  (unless (server-running-p)
    (server-start))
#+end_src

** Look and Feel

*** Theme

=ef-themes= are modern and readable. =ef-cherie= is the default. Load the
theme only in a graphical frame. Under =emacs -nw= (e.g. in tmux over
Ghostty) leave the faces unset so the terminal's own colours show through.

#+begin_src emacs-lisp :tangle "sns-emacs-modules/sns-look-and-feel.el" :mkdirp yes
;;; -*- lexical-binding: t -*-
(use-package ef-themes
  :ensure t
  :config
  (when (display-graphic-p)
    (load-theme 'ef-cherie t)))
#+end_src

*** Fonts and font resizing
Use Maple Mono NF at 15pt if installed; otherwise fall back to the system monospace font. Matches the font configured in Ghostty so the look stays consistent across editor and terminal. Only meaningful in a graphical frame: in a terminal the font is Ghostty's, so skip it under =emacs -nw=.

#+begin_src emacs-lisp :tangle "sns-emacs-modules/sns-look-and-feel.el" :mkdirp yes
(when (display-graphic-p)
  (if (member "Maple Mono NF" (font-family-list))
      (set-face-attribute 'default nil :font "Maple Mono NF-15")
    (set-face-attribute 'default nil :family "monospace" :height 150)))
#+end_src

**** Capture template

Ahrens-flavoured atomic notes. The title is the *claim*, not the
topic. =KIND= drives review: =permanent= (your idea, in your own
words, earns its place), =literature= (somebody else's idea you've
restated), =fleeting= (rough capture for later distillation).
=SOURCE= is mandatory for literature notes; left empty otherwise.
=CREATED= is stamped automatically.

#+begin_src emacs-lisp :tangle "sns-emacs-modules/sns-knowledge-base.el"
  (setq org-roam-capture-templates
        '(("p" "permanent" plain
           "%?"
           :target (file+head
                    "%<%Y%m%d%H%M%S>-${slug}.org"
                    ":PROPERTIES:\n:ID:       ${id}\n:CREATED:  %U\n:KIND:     permanent\n:END:\n#+title: ${title}\n\n")
           :unnarrowed t)
          ("l" "literature" plain
           "%?"
           :target (file+head
                    "%<%Y%m%d%H%M%S>-${slug}.org"
                    ":PROPERTIES:\n:ID:       ${id}\n:CREATED:  %U\n:KIND:     literature\n:SOURCE:   %^{Source}\n:END:\n#+title: ${title}\n\n")
           :unnarrowed t)
          ("f" "fleeting" plain
           "%?"
           :target (file+head
                    "%<%Y%m%d%H%M%S>-${slug}.org"
                    ":PROPERTIES:\n:ID:       ${id}\n:CREATED:  %U\n:KIND:     fleeting\n:END:\n#+title: ${title}\n\n")
           :unnarrowed t)))
#+end_src

*** Find and show double spaces

A minor mode that highlights double spaces, to help me break the habit of inserting them after full stops.

#+begin_src emacs-lisp :tangle "sns-emacs-modules/sns-technical-writing.el"
(define-minor-mode highlight-double-spaces-mode
  "Minor mode for highlighting double spaces using theme-defined colors."
  :lighter " DSpaces"
  :global nil
  :init-value nil
  (if highlight-double-spaces-mode
      (progn
        (font-lock-add-keywords nil '(("  +" 0 'highlight t)))
        (font-lock-flush)
        (font-lock-ensure))
    (progn
      (font-lock-remove-keywords nil '(("  +" 0 'highlight t)))
      (font-lock-flush)
      (font-lock-ensure))))
#+end_src

*** Presentations with reveal.js

=org-re-reveal= exports an org file to a [[https://revealjs.com/][reveal.js]] HTML slide deck
(dispatch with =C-c C-e v v=, or =C-c C-e v b= to open in the browser).
Each top-level heading becomes a slide; sub-headings nest as vertical slides.
=htmlize= gives syntax-highlighted code blocks in the exported HTML.

We use =org-re-reveal= rather than the older =ox-reveal=: ox-reveal has been
unmaintained since 2022 and its headline transcoder silently returns nil under
Org 9.8, so exports collapse to just the title slide. org-re-reveal is the
maintained fork and works with current Org.

The reveal.js library is pulled from a CDN so decks work anywhere without a
local install. Any deck can override these with =#+REVEAL_ROOT=, =#+REVEAL_THEME=,
etc. in its header.

#+begin_src emacs-lisp :tangle "sns-emacs-modules/sns-technical-writing.el"
  (use-package org-re-reveal
    :ensure t
    :after org
    :config
    (setq org-re-reveal-root "https://cdn.jsdelivr.net/npm/reveal.js@4"
          org-re-reveal-revealjs-version "4"
          org-re-reveal-theme "moon"
          org-re-reveal-transition "slide"
          org-re-reveal-mathjax t))
  (use-package htmlize
    :ensure t)
#+end_src

** Feeds

Feeds arrive as local news via Gnus. The =linkedin-miner= project scrapes
tracked authors' posts (plus the readable text of any article they link
to) and writes each as an RFC 5536 article into an =nnml= tree under
=~/Mail/linkedin/<author>/=, one newsgroup per author. Gnus reads that tree
directly: no NNTP server, no browser, no LinkedIn. Threading, scoring, and
killfiles come for free.

Open Gnus with =M-x gnus=. If the =linkedin.*= groups don't appear, press
=g= to rescan; new groups may need =A A= (list all) then subscribe, or the
group topic to be shown with =L=.

#+begin_src emacs-lisp :tangle "sns-emacs-modules/sns-feeds.el" :mkdirp yes
;;; -*- lexical-binding: t -*-
  ;; Gnus reads the linkedin-miner nnml tree as local news.
  (use-package gnus
    :ensure nil                      ; built in
    :bind ("C-x w" . gnus)
    :config
    ;; Primary select method: local nnml mail spool at ~/Mail.
    (setq gnus-select-method '(nnml "")
          nnml-directory "~/Mail"
          gnus-secondary-select-methods nil)
    ;; Show groups even with zero unread, and surface newly-created groups
    ;; without ceremony (the scraper adds a group per new author).
    (setq gnus-permanently-visible-groups "linkedin\\."
          gnus-save-killed-list nil
          gnus-check-new-newsgroups 'ask-server)
    ;; Read the plain-text article bodies comfortably: wrap and fill for prose.
    (add-hook 'gnus-article-mode-hook
              (lambda ()
                (setq-local fill-column 80)

*** Rename file and buffer
#+begin_src emacs-lisp :tangle "sns-lisp/sns-functions.el" :mkdirp yes
  (defun sns-rename-file-and-buffer ()
    "Rename the file visited by the current buffer and update the buffer to visit the renamed file.
  Prompts for a new name and handles version-controlled files appropriately. If the buffer is not
  visiting a file, signals an error. Ensures the buffer is updated after renaming."
    (interactive)
    (let ((filename (buffer-file-name)))
      (unless (and filename (file-exists-p filename))
	(user-error "Buffer is not visiting a file"))
      (let ((new-name (read-file-name "New name: " filename)))
	(if (vc-backend filename)
	    ;; Use version control rename if the file is under version control.
	    (vc-rename-file filename new-name)
	  ;; Rename file normally and update buffer.
	  (progn
	    (rename-file filename new-name t)
	    (set-visited-file-name new-name t t))))))
#+end_src

*** Reader view for org mode

A toggle for prose-friendly display: wider margins, extra line spacing, no line numbers. Useful when writing in the journal.

#+begin_src emacs-lisp :tangle "sns-lisp/sns-functions.el" :mkdirp yes
(use-package visual-fill-column
  :ensure t
  :config
  (setq-default visual-fill-column-center-text t)

  (defun sns-toggle-readable-view ()
    "Toggle a 'readable' view with centered text, wider margins, and hidden line numbers."
    (interactive)
    (if (bound-and-true-p visual-fill-column-mode)
        (progn
          (visual-fill-column-mode -1)
          (setq line-spacing nil)
          (display-line-numbers-mode 1)
          (message "Readable view disabled"))
      (progn
        (setq visual-fill-column-width 80)
        (setq visual-fill-column-center-text t)
        (setq line-spacing 0.3)
        (display-line-numbers-mode -1)
        (visual-fill-column-mode 1)
        (message "Readable view enabled")))))
#+end_src
